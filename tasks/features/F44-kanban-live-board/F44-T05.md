### Pipeline-to-column mapping
```js
function pipelineToColumn(p) {
  if (!p) return 'backlog';
  if (p.exit_code !== null && p.exit_code !== 0) return 'failed';
  if (p.status === 'done') return 'done';
  const reviewStages = ['techlead', 'qa', 'post_review'];
  if (reviewStages.includes(p.stage)) return 'review';
  if (p.status === 'running' || p.stage?.startsWith('wave')) return 'in_progress';
  if (p.status === 'planned' || p.stage === 'handler_gate') return 'planned';
  return 'backlog';
}
```

### Progress calculation
```js
function pipelineProgress(p) {
  if (!p) return 0;
  const completed = p.stages_completed?.length || 0;
  const total = completed + (p.stages_remaining?.length || 0);
  return total > 0 ? Math.round((completed / total) * 100) : 0;
}
```
