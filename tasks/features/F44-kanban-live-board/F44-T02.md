## Objective

responses into board-ready data structures. Embedded in the HTML file.

- Transform functions: API response → column-grouped data
- Event system: notify UI components on data update

### Data Transform Functions

```js
// War Room: group cards by status into columns
function groupWarroomCards(cards) {
  const columns = { incoming: [], triaging: [], needs_input: [],
                    promoted: [], deferred: [], archived: [] };
  cards.forEach(card => {
    const col = card.overlord_assessment?.recommendation === 'needs_input'
      ? 'needs_input' : card.status;
    if (columns[col]) columns[col].push(card);
  });
  return columns;
}

// Standard: derive column from pipeline state
function deriveFeatureColumn(pipeline) {
  if (!pipeline) return 'backlog';
  if (pipeline.exit_code && pipeline.exit_code !== 0) return 'failed';
  if (pipeline.status === 'done') return 'done';
  if (['techlead', 'qa', 'post_review'].includes(pipeline.stage)) return 'review';
  if (pipeline.status === 'running') return 'in_progress';
  return 'planned';
}
```
